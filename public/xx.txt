// title
title Inventory Management Data Model (V9 - No Categories)

// define tables

// -- Core Product & Classification Tables --
units_of_measure [icon: ruler, color: indigo]{
  id string pk
  name string
  abbreviation string
}

products [icon: package, color: yellow]{
  id string pk
  name string
  sku string
  unitOfMeasureId string
  createdAt datetime
  updatedAt datetime
}

product_variants [icon: layers, color: orange]{
  id string pk
  productId string
  name string
  attributes object
  sku string
  sellingPrice number
  createdAt datetime
  updatedAt datetime
}

// -- People Tables --
suppliers [icon: truck, color: blue]{
  id string pk
  name string
  contactName string
  email string
  phone string
  address string
  createdAt datetime
}

customers [icon: user, color: teal]{
  id string pk
  name string
  email string
  phone string
  address string
  createdAt datetime
}

// -- Core Inventory & Transaction Tables --
inventory [icon: database, color: lightblue]{
  id string pk
  productVariantId string
  quantity number
  minStockLevel number
  maxStockLevel number
  lastUpdated datetime
}

inventory_transactions [icon: repeat, color: pink]{
  id string pk
  productVariantId string
  transactionType string // purchase, sale, return, adjustment
  quantity number
  purchaseOrderItemId string
  salesOrderItemId string
  notes string
  createdAt datetime
}

// -- Purchasing Tables (Purchase Bills) --
purchase_orders [icon: clipboard, color: brown]{
  id string pk
  supplierId string
  orderDate datetime
  status string // pending, received, cancelled
  totalAmount number
  referenceNumber string
  createdAt datetime
}

purchase_order_items [icon: list, color: brown]{
  id string pk
  purchaseOrderId string
  productVariantId string
  quantity number
  unitCost number
  totalCost number
}

// -- Sales Tables (Sales Bills) --
sales_orders [icon: bar-chart-2, color: green]{
  id string pk
  customerId string
  orderDate datetime
  status string // completed, pending, cancelled
  totalAmount number
  createdAt datetime
  updatedAt datetime
}

sales_order_items [icon: chevrons-right, color: green]{
  id string pk
  salesOrderId string
  productVariantId string
  quantity number
  unitPrice number
  totalPrice number
}

// -- Financial & Ledger Tables --
customer_payments [icon: credit-card, color: orange] {
  id string pk
  salesOrderId string
  customerId string
  amount number
  paymentDate datetime
  method string // cash, card, online
}

supplier_payments [icon: chevrons-up, color: purple] {
    id string pk
    supplierId string
    amount number
    paymentDate datetime
    method string
    notes string
}

customer_ledger [icon: book-user, color: teal]{
    id string pk
    customerId string
    transactionType string // sale, payment, refund
    debit number // Amount customer owes you increases
    credit number // Amount customer owes you decreases
    balance number // The running balance after this transaction
    referenceId string // e.g., salesOrderId or paymentId
    createdAt datetime
}

supplier_ledger [icon: book-marked, color: blue]{
    id string pk
    supplierId string
    transactionType string // purchase, payment
    debit number // Amount you owe supplier decreases
    credit number // Amount you owe supplier increases
    balance number // The running balance after this transaction
    referenceId string // e.g., purchaseOrderId or paymentId
    createdAt datetime
}

// define relationships
// Product Structure Relationships
products.unitOfMeasureId > units_of_measure.id
products.id > product_variants.productId

// Inventory Relationships
product_variants.id > inventory.productVariantId
product_variants.id > inventory_transactions.productVariantId

// Purchasing Relationships
suppliers.id > purchase_orders.supplierId
purchase_orders.id > purchase_order_items.purchaseOrderId
product_variants.id > purchase_order_items.productVariantId
purchase_order_items.id > inventory_transactions.purchaseOrderItemId
suppliers.id > supplier_payments.supplierId
suppliers.id > supplier_ledger.supplierId
purchase_orders.id > supplier_ledger.referenceId
supplier_payments.id > supplier_ledger.referenceId

// Sales Relationships
customers.id > sales_orders.customerId
sales_orders.id > sales_order_items.salesOrderId
product_variants.id > sales_order_items.productVariantId
sales_order_items.id > inventory_transactions.salesOrderItemId

// Customer Payment & Ledger Relationships
customers.id > customer_payments.customerId
sales_orders.id > customer_payments.salesOrderId
customers.id > customer_ledger.customerId
sales_orders.id > customer_ledger.referenceId
customer_payments.id > customer_ledger.referenceId
